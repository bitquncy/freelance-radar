# ----------------------------------------
# 1. ПОДГОТОВКА GIT РЕПОЗИТОРИЯ
# ----------------------------------------

cd C:\Users\Пользователь\Desktop\freelance-radar

# Проверка статуса
git status

# Если git не инициализирован:
git init

# Добавить все файлы
git add .

# Сделать коммит
git commit -m "Initial commit: FreelanceRadar bot ready for Railway deployment"

# ----------------------------------------
# 2. СОЗДАНИЕ GITHUB РЕПОЗИТОРИЯ
# ----------------------------------------

# Откройте в браузере:
# https://github.com/new

# Создайте репозиторий с именем: freelance-radar
# Visibility: Private
# НЕ выбирайте "Initialize with README"

# ----------------------------------------
# 3. ПОДКЛЮЧЕНИЕ К GITHUB
# ----------------------------------------

# ЗАМЕНИТЕ YOUR_USERNAME на ваш GitHub username!
git remote add origin https://github.com/YOUR_USERNAME/freelance-radar.git

# Проверка remote
git remote -v

# Переименовать ветку в main
git branch -M main

# Отправить код на GitHub
git push -u origin main

# ----------------------------------------
# 4. НАСТРОЙКА RAILWAY
# ----------------------------------------

# Откройте в браузере:
# https://railway.app/

# 1. Login with GitHub
# 2. New Project
# 3. Deploy from GitHub repo
# 4. Выберите: freelance-radar

# ----------------------------------------
# 5. RAILWAY VOLUME (ВАЖНО!)
# ----------------------------------------

# В Railway проекте:
# Settings → Volumes → + Add Volume
# Mount Path: /app/data
# Size: 1 GB

# ----------------------------------------
# 6. RAILWAY VARIABLES
# ----------------------------------------

# Variables → + New Variable

# Добавьте (замените значения на реальные!):
